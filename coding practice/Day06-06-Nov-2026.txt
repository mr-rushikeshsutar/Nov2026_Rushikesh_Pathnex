    agent any
    stages{
        stage('checkout'){
            steps{
                git branch 'main', url:'https://github.com/Pathnex/sample-java-app.git'
            }
        }
        stage('build'){
            steps{
                sh '''
                mvn clean compile
                echo "compile failed; exit code1;"
                '''
            }
        }
        stage('test'){
            steps{
                sh '''
                mvn test 
                echo "test failed; exit code1;"
                '''
            }
        }
    }
}
