  hosts: all

  - name: install nginx
    yum:
    name: nginx
    state: present

#kubernetes Deployment-2replicas 

apiVersion: apps/v1
kind: Deployment
metadata:
 name: pathnex
 namespace: dev
spec:
 replicas: 2 
 selector:
  matchLabels:
   app: pathnex

 template:
  metadata:
   labels:
     app: pathnex

  spec:
   containers:
    - name: pathnex
      image: nginx
      ports:
        containerPort: 80

##completed in 4 min 1 sec

#jenkins-Parameterized

pipeline{
 agent any
 parameters{
 string(name: 'Env', defaultValue: 'dev',
        description: 'deploy in dev') 
}
 environment{
 INSTITUTE_NAME = "Pathnex"
}
 stages{
 stage('checkout'){
   steps{
  git branch 'main', url: 'https://github.com/Pathnex/sample-java-app.git'
}
}
 stage('deploy'){
  steps{
  sh 'echo "deployed to $Env by $INSTITUTE_NAME"' 
}
}
}
}
##completed in 8 min 2 sec
